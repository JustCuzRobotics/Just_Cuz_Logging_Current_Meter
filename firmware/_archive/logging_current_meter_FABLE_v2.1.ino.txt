/* ==========================================================================
 * logging_current_meter_FABLE.ino — v2.1 (2026-09-08)
 *
 * Full touchscreen UI for the Logging Current Meter (Rev A), rebuilt from
 * scratch on the touch architecture proven in FABLE_DEV_TEST_SCREEN v1.1.
 * Supersedes logging_current_meter_ui.ino v1.7 (kept as reference).
 *
 * Target: Waveshare RP2040-Zero, arduino-pico (earlephilhower) core.
 * Panel:  lcdwiki MSP3526 3.5" 480x320 IPS, ST7796 + FT6336U, shared FPC.
 * ZERO libraries beyond the core — raw-SPI ST7796 driver at 40 MHz and a
 * raw-I2C FT6336U driver, every bus byte visible in this file.
 *
 * --------------------------------------------------------------------------
 * THE CONTRACT — read before adding features (why touch stays reliable)
 *
 *   CORE 1 is the instrumentation core: it owns the FT6336U (I2C1) and the
 *   ADC, exclusively, and NOTHING else. Its loop:
 *     - polls the touch controller at a fixed TOUCH_HZ (200 Hz), FT6336U in
 *       POLLING mode (0xA4=0x00; CTP_INT is only *counted*, nothing waits
 *       on it), runs the press/release state machine, publishes the latest
 *       sample via a seqlock struct and DOWN/UP events via the inter-core
 *       FIFO;
 *     - runs the 13.158 ms ADC sample tick. The 4-channel oversampled burst
 *       takes ~5 ms, so touchService() is called BETWEEN channels — worst
 *       case touch jitter is one channel (~1.3 ms), invisible to a finger.
 *
 *   CORE 0 owns the display and all UI logic. It drains the touch FIFO
 *   every pass of loop() and repaints incrementally.
 *
 *   RULES:
 *     1. Core 1 must never block, print, or draw. Touch poll + ADC burst +
 *        bookkeeping only.
 *     2. Core 0 may be arbitrarily slow — a slow frame delays *reaction*,
 *        never *registration* (events queue in the FIFO; overflow is a
 *        counted glitch, not a silent drop).
 *     3. Never fillScreen()/full-frame repaint per tick. Chrome paints once
 *        per screen entry; per-tick redraw only what changed (drawField
 *        caches). Toast expiry repaints chrome — that's the one sanctioned
 *        full repaint outside navigation.
 *     4. Core 0 never touches the ADC, core 1 never touches SPI/display.
 *
 * --------------------------------------------------------------------------
 * TOUCH TARGETS (new in v2.0): every button has TWO rects — `vis` (what is
 * drawn) and `hit` (what registers, MUCH larger, bleeding into dead space
 * and screen edges). Hit rects are hand-assigned per button, replacing the
 * old uniform TOUCH_HIT_SLOP. checkTargetOverlaps() verifies at boot that
 * no two hit rects on the same screen overlap (serial warning if so), and
 * DEV MODE draws the real hit outlines of the four historically
 * troublesome targets.
 *
 * FONTS: Russo One (SIL OFL) pre-rasterized to 1-bit bitmaps by
 * make_fonts.py (in this folder): RUSSO16/RUSSO22 proportional for titles,
 * tile and action-button labels; RUSSOBIG (24 px, tabular/monospace digits
 * + E R C) for the big live values, so cached fixed-width field erasing
 * stays exact. The verified 5x7 handles all dense text (UPPERCASE-ONLY:
 * lowercase input is auto-uppercased).
 *
 * Version history
 *   v2.1  2026-09-08  Darker palette (box fills pushed deep so they recede on
 *                     this driver, matching the v1.7/Arduino_GFX look); V/T
 *                     show/hide chips fixed — toggling now repaints the plot
 *                     so a hidden trace disappears, and chip on/off states are
 *                     visually distinct (colored-on / grayed-off).
 *   v2.0  2026-09-08  Fresh rebuild: dual-core touch architecture from
 *                     FABLE_DEV_TEST_SCREEN v1.1 (core-1 200 Hz poller,
 *                     FIFO events, polling mode), raw 40 MHz ST7796 driver,
 *                     Russo One typography, per-button enlarged hit rects.
 *                     All v1.7 functionality preserved: Home/Live/Graph/
 *                     Calibrate/Dev screens, energy + 2/3-min marks, peaks,
 *                     tare, autoscaled 5 s graph, toasts, read-only cal.
 *
 * Carry-over hard lessons honored from v1.7 (do not regress):
 *   - Manual forward declarations for every function taking a struct/enum
 *     defined in this file (Arduino auto-prototypes land above the types).
 *   - uint16_t loop counters anywhere GRAPH_COLS (380) is the bound.
 *   - forceClear=true after ANY full repaint (screen entry, toast expiry)
 *     or cached fields stay blank.
 *   - SD_CS parked HIGH as the first GPIO touched (shares the FPC's MISO).
 *   - Backlight solid HIGH, never PWM (V5_SENSE bandwidth, DESIGN.md 4.2).
 *   - FT6336U: 0x86=0x00 forbids monitor mode (factory default silently
 *     drops scan rate after ~30 s idle); 0x88=0x04 fastest scan period.
 *   - Peak ordering: Vsag stored before peak mW; ring head advanced after
 *     data; temp fault carries last good ring sample forward.
 * ========================================================================*/

static inline void lcdCsLow()  { digitalWrite(PIN_LCD_CS, LOW);  }
static inline void lcdCsHigh() { digitalWrite(PIN_LCD_CS, HIGH); }
static inline void lcdDcCmd()  { digitalWrite(PIN_LCD_RS, LOW);  }
static inline void lcdDcData() { digitalWrite(PIN_LCD_RS, HIGH); }

static inline void spiBegin() {
  SPI.beginTransaction(SPISettings(SPI_HZ, MSBFIRST, SPI_MODE0));
  lcdCsLow();
}
static inline void spiEnd() { lcdCsHigh(); SPI.endTransaction(); }

static void lcdCmd(uint8_t c) { lcdDcCmd(); SPI.transfer(c); lcdDcData(); }
static void lcdCmdN(uint8_t c, const uint8_t *d, size_t n) {
  lcdCmd(c);
  if (n) SPI.transfer(d, nullptr, n);
}
static void lcdCmd1(uint8_t c, uint8_t a) { uint8_t d[1] = {a}; lcdCmdN(c, d, 1); }

static void lcdBegin() {
  pinMode(PIN_LCD_CS,  OUTPUT); digitalWrite(PIN_LCD_CS, HIGH);
  pinMode(PIN_LCD_RS,  OUTPUT); digitalWrite(PIN_LCD_RS, HIGH);
  pinMode(PIN_LCD_RST, OUTPUT);

  SPI.setSCK(PIN_SCK);
  SPI.setTX(PIN_MOSI);
  SPI.setRX(PIN_MISO);
  SPI.begin();

  digitalWrite(PIN_LCD_RST, HIGH); delay(10);
  digitalWrite(PIN_LCD_RST, LOW);  delay(20);
  digitalWrite(PIN_LCD_RST, HIGH); delay(120);

  spiBegin();
  lcdCmd(0x01); delay(120);                          /* SWRESET  */
  lcdCmd(0x11); delay(120);                          /* SLPOUT   */
  lcdCmd1(0xF0, 0xC3);
  lcdCmd1(0xF0, 0x96);
  lcdCmd1(0x36, LCD_MADCTL);
  lcdCmd1(0x3A, 0x55);                               /* 16-bit   */
  lcdCmd1(0xB4, 0x01);
  { const uint8_t d[] = {0x80, 0x02, 0x3B}; lcdCmdN(0xB6, d, sizeof d); }
  { const uint8_t d[] = {0x40, 0x8A, 0x00, 0x00, 0x29, 0x19, 0xA5, 0x33};
    lcdCmdN(0xE8, d, sizeof d); }
  lcdCmd1(0xC1, 0x06);
  lcdCmd1(0xC2, 0xA7);
  lcdCmd1(0xC5, 0x18);
  delay(120);
  { const uint8_t d[] = {0xF0, 0x09, 0x0B, 0x06, 0x04, 0x15, 0x2F,
                         0x54, 0x42, 0x3C, 0x17, 0x14, 0x18, 0x1B};
    lcdCmdN(0xE0, d, sizeof d); }
  { const uint8_t d[] = {0xE0, 0x09, 0x0B, 0x06, 0x04, 0x03, 0x2B,
                         0x43, 0x42, 0x3B, 0x16, 0x14, 0x17, 0x1B};
    lcdCmdN(0xE1, d, sizeof d); }
  delay(120);
  lcdCmd1(0xF0, 0x3C);
  lcdCmd1(0xF0, 0x69);
  delay(120);
#if LCD_INVERT
  lcdCmd(0x21);
#else
  lcdCmd(0x20);
#endif
  lcdCmd(0x29); delay(20);                           /* DISPON   */
  spiEnd();
}

static void lcdWindow(int16_t x, int16_t y, int16_t w, int16_t h) {
  uint8_t d[4];
  d[0] = x >> 8; d[1] = x & 0xFF;
  d[2] = (x + w - 1) >> 8; d[3] = (x + w - 1) & 0xFF;
  lcdCmdN(0x2A, d, 4);
  d[0] = y >> 8; d[1] = y & 0xFF;
  d[2] = (y + h - 1) >> 8; d[3] = (y + h - 1) & 0xFF;
  lcdCmdN(0x2B, d, 4);
  lcdCmd(0x2C);
}

static void fillRect(int16_t x, int16_t y, int16_t w, int16_t h, uint16_t c) {
  if (x < 0) { w += x; x = 0; }
  if (y < 0) { h += y; y = 0; }
  if (x + w > LCD_W) w = LCD_W - x;
  if (y + h > LCD_H) h = LCD_H - y;
  if (w <= 0 || h <= 0) return;
  uint16_t be = __builtin_bswap16(c);
  for (int16_t i = 0; i < w; i++) s_lineBuf[i] = be;
  spiBegin();
  lcdWindow(x, y, w, h);
  for (int16_t row = 0; row < h; row++)
    SPI.transfer((uint8_t *)s_lineBuf, nullptr, (size_t)w * 2);
  spiEnd();
}

static void fillScreen(uint16_t c)                { fillRect(0, 0, LCD_W, LCD_H, c); }
static void drawFastHLine(int16_t x, int16_t y, int16_t w, uint16_t c) { fillRect(x, y, w, 1, c); }
static void drawFastVLine(int16_t x, int16_t y, int16_t h, uint16_t c) { fillRect(x, y, 1, h, c); }
static void drawPixel(int16_t x, int16_t y, uint16_t c)                { fillRect(x, y, 1, 1, c); }
static void drawRect(int16_t x, int16_t y, int16_t w, int16_t h, uint16_t c) {
  drawFastHLine(x, y, w, c);
  drawFastHLine(x, y + h - 1, w, c);
  drawFastVLine(x, y, h, c);
  drawFastVLine(x + w - 1, y, h, c);
}

static void drawLine(int16_t x0, int16_t y0, int16_t x1, int16_t y1, uint16_t c) {
  int16_t dx = abs(x1 - x0), sx = x0 < x1 ? 1 : -1;
  int16_t dy = -abs(y1 - y0), sy = y0 < y1 ? 1 : -1;
  int16_t err = dx + dy;
  for (;;) {
    drawPixel(x0, y0, c);
    if (x0 == x1 && y0 == y1) break;
    int16_t e2 = 2 * err;
    if (e2 >= dy) { err += dy; x0 += sx; }
    if (e2 <= dx) { err += dx; y0 += sy; }
  }
}

/* Midpoint circle — quadrant helpers in the Adafruit corner convention:
 * corner bits 1,2,4,8 = NW, NE, SE, SW arcs respectively. */
static void drawCircleHelper(int16_t x0, int16_t y0, int16_t r, uint8_t corner, uint16_t c) {
  int16_t f = 1 - r, ddF_x = 1, ddF_y = -2 * r, x = 0, y = r;
  while (x < y) {
    if (f >= 0) { y--; ddF_y += 2; f += ddF_y; }
    x++; ddF_x += 2; f += ddF_x;
    if (corner & 0x4) { drawPixel(x0 + x, y0 + y, c); drawPixel(x0 + y, y0 + x, c); }
    if (corner & 0x2) { drawPixel(x0 + x, y0 - y, c); drawPixel(x0 + y, y0 - x, c); }
    if (corner & 0x8) { drawPixel(x0 - y, y0 + x, c); drawPixel(x0 - x, y0 + y, c); }
    if (corner & 0x1) { drawPixel(x0 - y, y0 - x, c); drawPixel(x0 - x, y0 - y, c); }
  }
}
static void fillCircleHelper(int16_t x0, int16_t y0, int16_t r, uint8_t sides,
                             int16_t delta, uint16_t c) {
  int16_t f = 1 - r, ddF_x = 1, ddF_y = -2 * r, x = 0, y = r;
  int16_t px = x, py = y;
  delta++;
  while (x < y) {
    if (f >= 0) { y--; ddF_y += 2; f += ddF_y; }
    x++; ddF_x += 2; f += ddF_x;
    if (x < y + 1) {
      if (sides & 1) drawFastVLine(x0 + x, y0 - y, 2 * y + delta, c);
      if (sides & 2) drawFastVLine(x0 - x, y0 - y, 2 * y + delta, c);
    }
    if (y != py) {
      if (sides & 1) drawFastVLine(x0 + py, y0 - px, 2 * px + delta, c);
      if (sides & 2) drawFastVLine(x0 - py, y0 - px, 2 * px + delta, c);
      py = y;
    }
    px = x;
  }
}
static void drawCircle(int16_t x0, int16_t y0, int16_t r, uint16_t c) {
  drawPixel(x0, y0 + r, c); drawPixel(x0, y0 - r, c);
  drawPixel(x0 + r, y0, c); drawPixel(x0 - r, y0, c);
  drawCircleHelper(x0, y0, r, 0xF, c);
}
static void fillCircle(int16_t x0, int16_t y0, int16_t r, uint16_t c) {
  drawFastVLine(x0, y0 - r, 2 * r + 1, c);
  fillCircleHelper(x0, y0, r, 3, 0, c);
}
static void drawRoundRect(int16_t x, int16_t y, int16_t w, int16_t h, int16_t r, uint16_t c) {
  int16_t maxR = ((w < h) ? w : h) / 2;
  if (r > maxR) r = maxR;
  drawFastHLine(x + r, y, w - 2 * r, c);
  drawFastHLine(x + r, y + h - 1, w - 2 * r, c);
  drawFastVLine(x, y + r, h - 2 * r, c);
  drawFastVLine(x + w - 1, y + r, h - 2 * r, c);
  drawCircleHelper(x + r, y + r, r, 1, c);
  drawCircleHelper(x + w - r - 1, y + r, r, 2, c);
  drawCircleHelper(x + w - r - 1, y + h - r - 1, r, 4, c);
  drawCircleHelper(x + r, y + h - r - 1, r, 8, c);
}
static void fillRoundRect(int16_t x, int16_t y, int16_t w, int16_t h, int16_t r, uint16_t c) {
  int16_t maxR = ((w < h) ? w : h) / 2;
  if (r > maxR) r = maxR;
  fillRect(x + r, y, w - 2 * r, h, c);
  fillRect(x, y + r, r, h - 2 * r, c);
  fillRect(x + w - r, y + r, r, h - 2 * r, c);
  fillCircleHelper(x + r, y + r, r, 2, 0, c);
  fillCircleHelper(x + w - r - 1, y + r, r, 1, 0, c);
  fillCircleHelper(x + r, y + h - r - 1, r, 2, 0, c);   /* lower quadrants: */
  fillCircleHelper(x + w - r - 1, y + h - r - 1, r, 1, 0, c);
}

/* ---- 5x7 text (scalable, opaque cell, auto-uppercase) ---- */
static void drawText5(int16_t x, int16_t y, const char *s,
                      uint16_t fg, uint16_t bg, uint8_t scale) {
  spiBegin();
  uint16_t feBe = __builtin_bswap16(fg), bgBe = __builtin_bswap16(bg);
  for (; *s; s++, x += 6 * scale) {
    char ch = *s;
    if (ch >= 'a' && ch <= 'z') ch -= 32;
    uint8_t idx = (ch < 32 || ch > 95) ? ('?' - 32) : (ch - 32);
    const uint8_t *col = FONT5x7[idx];
    if (x + 6 * scale > LCD_W || y + 8 * scale > LCD_H || x < 0 || y < 0) continue;
    lcdWindow(x, y, 6 * scale, 8 * scale);
    for (uint8_t ry = 0; ry < 8; ry++) {
      for (uint8_t sy = 0; sy < scale; sy++) {
        uint16_t *p = s_lineBuf;
        for (uint8_t cx = 0; cx < 6; cx++) {
          bool on = (cx < 5) && (ry < 7) && ((col[cx] >> ry) & 1);
          uint16_t v = on ? feBe : bgBe;
          for (uint8_t sx = 0; sx < scale; sx++) *p++ = v;
        }
        SPI.transfer((uint8_t *)s_lineBuf, nullptr, (size_t)6 * scale * 2);
      }
    }
  }
  spiEnd();
}
static int16_t text5W(const char *s, uint8_t scale) {
  return (int16_t)strlen(s) * 6 * scale;
}

/* ---- Russo One text (variable width, opaque cell) ---- */
static uint8_t russoIdx(char ch) {
  if (ch >= 'a' && ch <= 'z') ch -= 32;
  if (ch < 32 || ch > 90) ch = ' ';
  return (uint8_t)(ch - 32);
}
static void drawRusso(const RussoFont &f, int16_t x, int16_t y,
                      const char *s, uint16_t fg, uint16_t bg) {
  spiBegin();
  uint16_t feBe = __builtin_bswap16(fg), bgBe = __builtin_bswap16(bg);
  for (; *s; s++) {
    uint8_t gi = russoIdx(*s);
    uint8_t gw = f.w[gi], adv = f.adv[gi];
    if (!adv) { gi = russoIdx('-'); gw = f.w[gi]; adv = f.adv[gi]; }  /* '-' exists in every set */
    if (!adv) continue;                        /* still unmapped: skip, no w=0 window */
    if (x + adv > LCD_W || y + f.h > LCD_H || x < 0 || y < 0) { x += adv; continue; }
    const uint8_t *gd = f.data + f.off[gi];
    lcdWindow(x, y, adv, f.h);
    for (uint8_t ry = 0; ry < f.h; ry++) {
      uint16_t *p = s_lineBuf;
      for (uint8_t cx = 0; cx < adv; cx++) {
        bool on = (cx < gw) && ((gd[cx * f.bpc + (ry >> 3)] >> (ry & 7)) & 1);
        *p++ = on ? feBe : bgBe;
      }
      SPI.transfer((uint8_t *)s_lineBuf, nullptr, (size_t)adv * 2);
    }
    x += adv;
  }
  spiEnd();
}
static int16_t russoTextW(const RussoFont &f, const char *s) {
  int16_t w = 0;
  for (; *s; s++) {
    uint8_t gi = russoIdx(*s);
    w += f.adv[gi] ? f.adv[gi] : f.adv[russoIdx('-')];
  }
  return w;
}
static void drawRussoCentered(const RussoFont &f, int16_t cx, int16_t y,
                              const char *s, uint16_t fg, uint16_t bg) {
  drawRusso(f, cx - russoTextW(f, s) / 2, y, s, fg, bg);
}

/* ---- cached fields ----
 * 5x7 field: fixed 6*scale advance, so erase width = maxChars cells.
 * Big field: RUSSOBIG tabular advance, same fixed-width reasoning.
 * cache==nullptr -> always redraw (v1.7 timer-field behavior). */
static bool drawField5(int16_t x, int16_t y, uint8_t maxChars, uint8_t scale,
                       uint16_t fg, uint16_t bg, const char *text, char *cache) {
  if (cache && strncmp(cache, text, maxChars) == 0) return false;
  if (cache) { strncpy(cache, text, maxChars); cache[maxChars] = 0; }
  fillRect(x, y, (int16_t)maxChars * 6 * scale, 8 * scale, bg);
  drawText5(x, y, text, fg, bg, scale);
  return true;
}
static bool drawFieldBig(int16_t x, int16_t y, uint8_t maxChars,
                         uint16_t fg, uint16_t bg, const char *text, char *cache) {
  char t[12];
  if (maxChars > 11) maxChars = 11;
  strncpy(t, text, maxChars); t[maxChars] = 0;   /* hard truncate: can never
                                                  * paint past the erase region */
  if (cache && strcmp(cache, t) == 0) return false;
  if (cache) strcpy(cache, t);
  fillRect(x, y, (int16_t)maxChars * RBIG_ADV, RUSSOBIG_H, bg);
  drawRusso(RBIG, x, y, t, fg, bg);
  return true;
}

/* ==========================================================================
 * TOUCH — FT6336U on I2C1, POLLING mode. Everything in this section except
 * takeSnap() runs on CORE 1. Proven engine from FABLE_DEV_TEST_SCREEN v1.1.
 * ========================================================================*/

/* --- shared state: core 1 writes, core 0 reads (seqlock) --- */
static volatile uint32_t g_seq = 0;
static volatile struct {
  uint16_t rawX, rawY;
  uint8_t  pts;
  bool     down;
  uint32_t tDownUs;
} g_touch;

/* --- diagnostic counters: core-1/ISR increment, core 0 reads. 32-bit
 *     aligned loads/stores are atomic on RP2040. --- */
static volatile uint32_t g_cntDown = 0, g_cntUp = 0;
static volatile uint32_t g_glRange = 0, g_glJump = 0, g_glCount = 0,
                         g_glDrop = 0, g_glOvf = 0, g_i2cErr = 0;
static volatile uint32_t g_intEdges = 0;
static volatile uint32_t g_touchHz = 0;
static volatile bool     g_ftOk = false;
static volatile bool     g_statReset = false;   /* core 0 sets, core 1 clears */

/* FIFO event packing: [31:28]=type, [23:12]=rawX, [11:0]=rawY */
#define EV_DOWN 1u
#define EV_UP   2u
static inline uint32_t evPack(uint32_t t, uint32_t x, uint32_t y) {
  return (t << 28) | ((x & 0xFFF) << 12) | (y & 0xFFF);
}

static void ctpIntIsr() { g_intEdges++; }   /* diagnostic only */

static bool ftWrite(uint8_t reg, uint8_t val) {
  Wire1.beginTransmission(FT_ADDR);
  Wire1.write(reg);
  Wire1.write(val);
  return Wire1.endTransmission() == 0;
}
static bool ftRead(uint8_t reg, uint8_t *buf, uint8_t n) {
  Wire1.beginTransmission(FT_ADDR);
  Wire1.write(reg);
  if (Wire1.endTransmission(false) != 0) return false;
  if (Wire1.requestFrom((uint8_t)FT_ADDR, n) != n) return false;
  for (uint8_t i = 0; i < n; i++) buf[i] = Wire1.read();
  return true;
}

/* One touch sample: burst-read 0x02..0x06, run the press/release state
 * machine, publish. ~120 us at 400 kHz. Called from the core-1 scheduler
 * AND between ADC channels (see sampleTick) so the ADC burst can't starve
 * it. */
static void touchSampleNow() {
  static bool     down = false;
  static uint8_t  emptyRun = 0;
  static uint16_t lastX = 0, lastY = 0;
  static uint32_t tDownUs = 0;
  static uint32_t sampleCount = 0, hzWindowMs = 0;

  uint8_t b[5];
  bool ok = ftRead(0x02, b, 5);
  sampleCount++;
  uint32_t nowMs = millis();
  if (nowMs - hzWindowMs >= 1000) { g_touchHz = sampleCount; sampleCount = 0; hzWindowMs = nowMs; }

  if (!ok) { g_i2cErr++; return; }

  uint8_t pts = b[0] & 0x0F;
  if (pts > 2) { g_glCount++; return; }

  if (pts == 0) {
    if (down) {
      if (++emptyRun >= RELEASE_CONFIRM) {
        down = false; emptyRun = 0;
        g_cntUp++;
        if (!rp2040.fifo.push_nb(evPack(EV_UP, lastX, lastY))) g_glOvf++;
      }
    } else emptyRun = 0;
  } else {
    uint16_t x = ((uint16_t)(b[1] & 0x0F) << 8) | b[2];
    uint16_t y = ((uint16_t)(b[3] & 0x0F) << 8) | b[4];
    if (x > 319 || y > 479) { g_glRange++; return; }
    if (down && emptyRun) g_glDrop++;   /* bridged single-sample dropout */
    emptyRun = 0;
    if (down) {
      int16_t dx = (int16_t)x - (int16_t)lastX;
      int16_t dy = (int16_t)y - (int16_t)lastY;
      if (dx < 0) dx = -dx;
      if (dy < 0) dy = -dy;
      if (dx > JUMP_PX || dy > JUMP_PX) g_glJump++;
    } else {
      down = true;
      tDownUs = micros();
      g_cntDown++;
      if (!rp2040.fifo.push_nb(evPack(EV_DOWN, x, y))) g_glOvf++;
    }
    lastX = x; lastY = y;
  }

  g_seq++;
  g_touch.rawX = lastX;
  g_touch.rawY = lastY;
  g_touch.pts  = pts;
  g_touch.down = down;
  g_touch.tDownUs = tDownUs;
  g_seq++;
}

/* Fixed-cadence gate for the touch poll; safe to call as often as you
 * like — it only samples when the next 1/TOUCH_HZ slot is due. */
static void touchService() {
  static uint32_t nextUs = 0;
  uint32_t now = micros();
  if (nextUs == 0) nextUs = now;
  if ((int32_t)(now - nextUs) < 0) return;
  nextUs += 1000000UL / TOUCH_HZ;
  if ((int32_t)(micros() - nextUs) > 0) nextUs = micros();  /* fell behind */
  touchSampleNow();
}

static void takeSnap(TouchSnap &s) {
  uint32_t s1, s2;
  do {
    s1 = g_seq;
    s.rawX = g_touch.rawX; s.rawY = g_touch.rawY;
    s.pts  = g_touch.pts;  s.down = g_touch.down;
    s.tDownUs = g_touch.tDownUs;
    s2 = g_seq;
  } while (s1 != s2 || (s1 & 1));
}

/* raw (portrait 320x480) -> screen (landscape 480x320); mapping confirmed
 * on hardware by FABLE_DEV_TEST_SCREEN v1.1's crosshair. */
static void mapTouch(uint16_t rx, uint16_t ry, int16_t &mx, int16_t &my) {
#if TOUCH_SWAP_XY
  mx = (int16_t)ry;  my = (int16_t)rx;
#else
  mx = (int16_t)rx;  my = (int16_t)ry;
#endif
#if TOUCH_INV_X
  mx = (LCD_W - 1) - mx;
#endif
#if TOUCH_INV_Y
  my = (LCD_H - 1) - my;
#endif
  if (mx < 0) mx = 0;
  if (mx >= LCD_W) mx = LCD_W - 1;
  if (my < 0) my = 0;
  if (my >= LCD_H) my = LCD_H - 1;
}

/* ==========================================================================
 * ADC + calibration math — verbatim from v1.7 / display_bringup.ino.
 * CORE 1 ONLY (core 0 never calls analogRead — see THE CONTRACT).
 * ========================================================================*/
static uint16_t adcAvg(uint8_t pin) {
  uint32_t acc = 0;
  for (uint16_t i = 0; i < ADC_AVG; i++) { acc += analogRead(pin); delayMicroseconds(20); }
  return (uint16_t)(acc / ADC_AVG);
}
static float packVolts(uint16_t raw) { return V_GAIN_CAL * (float)raw + V_OFFSET_CAL; }

/* Both VCC and VREF cancel out of this ratio — DESIGN.md 4.2 — which makes
 * the current reading independent of USB rail noise. */
static float iRatio(uint16_t rawI, uint16_t raw5) {
  if (raw5 < 100) return 0.0f;
  return (float)rawI / (float)raw5 * G2_OVER_G1;
}

/* NTC through RV1, B-parameter form. x>0.97 open, x<0.02 short -> fault. */
static bool tempFromRawC(uint16_t rawT, float *outC) {
  float x = rawT / ADC_MAX;
  if (x > 0.97f || x < 0.02f) return false;
  float rntc = RV1_OHMS * x / (1.0f - x);
  *outC = 1.0f / (1.0f / 298.15f + logf(rntc / NTC_R25) / NTC_B) - 273.15f;
  return true;
}

/* Fixed-point convention (v1.7, unchanged): centiamps/centivolts/centidegc
 * int16_t = value*100; milliwatts int32_t = W*1000. Float only for
 * calibration math + energy integration. Sentinel = thermistor fault. */
#define FIXED_INVALID  ((int16_t)-32768)

/* Cross-core shared state. Ownership rule (v1.7, unchanged): every field
 * written by core 1 only, read by core 0 only; aligned 16/32-bit stores
 * are atomic on RP2040. The cmd flags are the one core-0 write: core 0
 * only sets TRUE, core 1 only sets FALSE. */
struct SampleState {
  int16_t  centiamps, centivolts, centidegc;
  int32_t  milliwatts;
  float    energyWh, energyMah;
  bool     mark2Captured, mark3Captured;
  float    mark2Wh, mark2Mah, mark3Wh, mark3Mah;
  uint32_t runElapsedMs;
  int16_t  peakCentiamps, peakVAtPeakCentivolts, peakCentidegc;
  int32_t  peakMilliwatts;
  bool     tempFault;
};
volatile SampleState gState;

volatile bool gCmdTare = false;
volatile bool gCmdResetEnergyTimer = false;
volatile bool gCmdResetPeaks = false;
volatile bool gCoreReady = false;

/* Graph ring — core 1 writes, core 0 reads; head = NEWEST, advanced AFTER
 * the data store. 380 cols = plot width, 5 s window, 13.158 ms/tick. */
#define GRAPH_COLS   380
#define TICK_US      13158UL

int16_t gRingI[GRAPH_COLS], gRingV[GRAPH_COLS], gRingT[GRAPH_COLS];
volatile uint16_t gRingHead = 0;

static float gTareOffsetA = 0.0f;   /* runtime-only; never touches I_QUIESCENT_CAL */

/* One 13.158 ms ADC tick — v1.7's sampleTick verbatim, EXCEPT that
 * touchService() runs between the four channel bursts so the ~5 ms of
 * oversampling can never delay a touch sample by more than ~1.3 ms
 * (THE CONTRACT, rule 1). */
static void sampleTick() {
  uint16_t rawI = adcAvg(PIN_I_SENSE);
  touchService();
  uint16_t raw5 = adcAvg(PIN_V5_SENSE);
  touchService();
  uint16_t rawV = adcAvg(PIN_V_PACK);
  touchService();
  uint16_t rawT = adcAvg(PIN_T_SENSE);
  touchService();

  float ratio   = iRatio(rawI, raw5);
  float ampsRaw = (ratio - I_QUIESCENT_CAL) / I_SENS_CAL;

  /* Tare: zero the reading only, sanity-gated near the nominal ~0.1
   * quiescent ratio; never touches the baked calibration. */
  if (gCmdTare) {
    if (ratio > 0.05f && ratio < 0.15f) gTareOffsetA = -ampsRaw;
    gCmdTare = false;
  }
  float amps = ampsRaw + gTareOffsetA;
  if (amps < 0.0f) amps = 0.0f;

  float volts = packVolts(rawV);
  float tempC; bool tempOk = tempFromRawC(rawT, &tempC);
  float watts = volts * amps;

  int16_t centiamps  = (int16_t)lroundf(amps  * 100.0f);
  int16_t centivolts = (int16_t)lroundf(volts * 100.0f);
  int16_t centidegc  = tempOk ? (int16_t)lroundf(tempC * 100.0f) : FIXED_INVALID;
  int32_t milliwatts = (int32_t)lroundf(watts * 1000.0f);

  gState.centiamps  = centiamps;
  gState.centivolts = centivolts;
  gState.centidegc  = centidegc;
  gState.milliwatts = milliwatts;
  gState.tempFault  = !tempOk;

  /* Energy integration with MEASURED dt (not the nominal tick). */
  static uint32_t lastTickUs = 0;
  uint32_t nowUs = micros();
  uint32_t dtUs = lastTickUs ? (nowUs - lastTickUs) : (uint32_t)TICK_US;
  lastTickUs = nowUs;
  const float K_WH_PER_US = 1.0f / 3600000000.0f;
  gState.energyWh  += watts * (float)dtUs * K_WH_PER_US;
  gState.energyMah += amps * 1000.0f * (float)dtUs * K_WH_PER_US;
  gState.runElapsedMs += dtUs / 1000UL;

  /* Peaks. Vsag is written BEFORE peak mW (fixed order) so core 0 never
   * pairs a new peak power with a stale sag voltage. */
  if (centiamps > gState.peakCentiamps) gState.peakCentiamps = centiamps;
  if (milliwatts > gState.peakMilliwatts) {
    gState.peakVAtPeakCentivolts = centivolts;
    gState.peakMilliwatts = milliwatts;
  }
  if (tempOk && centidegc > gState.peakCentidegc) gState.peakCentidegc = centidegc;

  /* Estimated-energy freeze marks at 2 and 3 minutes elapsed. */
  if (!gState.mark2Captured && gState.runElapsedMs >= 120000UL) {
    gState.mark2Wh = gState.energyWh; gState.mark2Mah = gState.energyMah;
    gState.mark2Captured = true;
  }
  if (!gState.mark3Captured && gState.runElapsedMs >= 180000UL) {
    gState.mark3Wh = gState.energyWh; gState.mark3Mah = gState.energyMah;
    gState.mark3Captured = true;
  }

  /* Peak Reset: I/P/Vsag -> 0, T -> current reading (0 C isn't a floor). */
  if (gCmdResetPeaks) {
    gState.peakCentiamps = 0;
    gState.peakMilliwatts = 0;
    gState.peakVAtPeakCentivolts = 0;
    gState.peakCentidegc = tempOk ? centidegc : FIXED_INVALID;
    gCmdResetPeaks = false;
  }

  /* Energy Reset and Timer Reset are the SAME combined action. */
  if (gCmdResetEnergyTimer) {
    gState.energyWh = 0.0f; gState.energyMah = 0.0f;
    gState.runElapsedMs = 0;
    gState.mark2Captured = gState.mark3Captured = false;
    gCmdResetEnergyTimer = false;
  }

  /* Ring: head advances AFTER the store; temp fault carries last good T. */
  int16_t tForRing = tempOk ? centidegc : gRingT[gRingHead];
  uint16_t h = (gRingHead + 1) % GRAPH_COLS;
  gRingI[h] = centiamps;
  gRingV[h] = centivolts;
  gRingT[h] = tForRing;
  gRingHead = h;
}

/* ---------------- core 1 entry points ---------------- */
void setup1() {
  pinMode(PIN_CTP_RST, OUTPUT);
  pinMode(PIN_CTP_INT, INPUT_PULLUP);

  digitalWrite(PIN_CTP_RST, HIGH); delay(10);
  digitalWrite(PIN_CTP_RST, LOW);  delay(20);
  digitalWrite(PIN_CTP_RST, HIGH); delay(300);   /* FT boot time */

  Wire1.setSDA(PIN_CTP_SDA);
  Wire1.setSCL(PIN_CTP_SCL);
  Wire1.begin();
  Wire1.setClock(100000);            /* init writes at the conservative speed */
  delay(50);

  bool ok = true;
  ok &= ftWrite(0x00, 0x00);   /* DEV_MODE: normal                          */
  ok &= ftWrite(0xA5, 0x00);   /* P_ACTIVE: stay active                     */
  ok &= ftWrite(0x86, 0x00);   /* FORBID monitor mode — the one that matters:
                                * factory default silently drops the scan
                                * rate after ~30 s idle (v1.7 lesson)       */
  ok &= ftWrite(0x88, 0x04);   /* fastest scan period                       */
  ok &= ftWrite(0xA4, 0x00);   /* G_MODE: POLLING — INT is diagnostic only  */
  g_ftOk = ok;

  Wire1.setClock(TOUCH_I2C_HZ);
  delay(5);

  attachInterrupt(digitalPinToInterrupt(PIN_CTP_INT), ctpIntIsr, FALLING);

  while (!gCoreReady) { /* wait for core 0's display init */ }
}

void loop1() {
  if (g_statReset) {
    g_cntDown = g_cntUp = 0;
    g_glRange = g_glJump = g_glCount = g_glDrop = g_glOvf = g_i2cErr = 0;
    g_intEdges = 0;
    g_statReset = false;
  }

  touchService();

  /* ADC tick scheduler — v1.7's catch-up/resync pattern. */
  static uint32_t next = 0;
  if (!next) next = micros();
  if ((int32_t)(micros() - next) >= 0) {
    next += TICK_US;
    if ((int32_t)(micros() - next) > (int32_t)TICK_US * 4)
      next = micros() + TICK_US;
    sampleTick();
  }
}

/* ---- value formatters — verbatim from v1.7 ---- */
static void fmt3SigFig(int32_t whole, uint16_t frac100, char *buf, size_t n) {
  if (whole < 10)       snprintf(buf, n, "%ld.%02u", (long)whole, frac100);
  else if (whole < 100) snprintf(buf, n, "%ld.%u",   (long)whole, frac100 / 10);
  else                  snprintf(buf, n, "%ld",      (long)whole);
}
static void fmt3SigCenti(int16_t centi, char *buf, size_t n) {
  if (centi == FIXED_INVALID) { snprintf(buf, n, "ERR"); return; }
  if (centi < 0) centi = 0;   /* no pack: V_OFFSET_CAL reads ~-0.14 V — clamp,
                               * or fmt3SigFig's unsigned modulo prints garbage */
  fmt3SigFig(centi / 100, (uint16_t)(centi % 100), buf, n);
}
static void fmt3SigWatts(int32_t milliwatts, char *buf, size_t n) {
  if (milliwatts < 0) milliwatts = 0;   /* same clamp as fmt3SigCenti */
  fmt3SigFig(milliwatts / 1000, (uint16_t)((milliwatts % 1000) / 10), buf, n);
}
static void fmtDegC(int16_t centidegc, char *buf, size_t n) {
  if (centidegc == FIXED_INVALID) { snprintf(buf, n, "ERR"); return; }
  snprintf(buf, n, "%dC", centidegc / 100);
}
static void fmtTimer(uint32_t ms, char *buf, size_t n) {
  uint32_t totalSec = ms / 1000, m = totalSec / 60, s = totalSec % 60, msPart = ms % 1000;
  snprintf(buf, n, "%02lu:%02lu.%03lu", (unsigned long)m, (unsigned long)s, (unsigned long)msPart);
}
static void fmtMark(bool captured, float wh, float mah, char *buf, size_t n) {
  if (!captured) snprintf(buf, n, "---");
  else snprintf(buf, n, "%.2fWh %.0fmA", wh, mah);
}
static void fmtWhole(int16_t centi, char *buf, size_t n) {
  if (centi == FIXED_INVALID) { snprintf(buf, n, "ERR"); return; }
  snprintf(buf, n, "%d", centi / 100);
}
static void fmtAxisAmps(int16_t centi, char *buf, size_t n) {
  snprintf(buf, n, "%d.%d", centi / 100, (centi % 100) / 10);
}

/* ==========================================================================
 * TOUCH TARGETS — visual rect + ENLARGED hit rect per button (new in v2.0).
 *
 * Hit rects are hand-assigned: corner/edge buttons bleed all the way to
 * the screen edges; interior buttons take the dead space around them;
 * where two targets share a boundary the gap is split. Display-only boxes
 * (stat boxes, timer readouts, present/peak rows) are NOT targets, so hit
 * rects may freely cover them.
 * Verified non-overlapping per screen by checkTargetOverlaps() at boot.
 * ========================================================================*/
static bool inRect(const Rect &r, int16_t x, int16_t y) {
  return x >= r.x && x < r.x + r.w && y >= r.y && y < r.y + r.h;
}
static int8_t hitTestTargets(const Target *t, uint8_t n, int16_t x, int16_t y) {
  for (uint8_t i = 0; i < n; i++)
    if (inRect(t[i].hit, x, y)) return (int8_t)i;
  return -1;
}

/* Boot-time overlap check: any two hit rects overlapping on one screen is
 * a layout bug — warn on serial (DEV MODE also draws the real hit rects). */
static void overlapWarn(const Target *t, uint8_t n, const char *screen) {
  for (uint8_t a = 0; a < n; a++)
    for (uint8_t b = a + 1; b < n; b++) {
      const Rect &p = t[a].hit, &q = t[b].hit;
      bool sep = (p.x + p.w <= q.x) || (q.x + q.w <= p.x) ||
                 (p.y + p.h <= q.y) || (q.y + q.h <= p.y);
      if (!sep)
        Serial.printf("[LAYOUT BUG] %s: hit rects %u and %u overlap!\n", screen, a, b);
    }
}
static void checkTargetOverlaps() {
  overlapWarn(HOME_T, HOME_N, "HOME");
  overlapWarn(LIVE_T, LIVE_BTN_N, "LIVE");
  overlapWarn(GRAPH_T, GRAPH_BTN_N, "GRAPH");
}

/* ==========================================================================
 * Toast — bottom-center, timed, non-blocking. No framebuffer: expiry
 * repaints the current screen's chrome (paintXOnce is idempotent; at
 * 40 MHz SPI a full repaint is ~60-80 ms, once, on expiry only).
 * ========================================================================*/
#define TOAST_MS  1600
static char gToastMsg[40];
static uint32_t gToastUntil = 0;

static void showToast(const char *msg) {
  strncpy(gToastMsg, msg, sizeof(gToastMsg) - 1);
  gToastMsg[sizeof(gToastMsg) - 1] = 0;
  gToastUntil = millis() + TOAST_MS;
  int16_t w = (int16_t)strlen(gToastMsg) * 6 + 16;
  int16_t x = (LCD_W - w) / 2, y = LCD_H - 38;
  fillRoundRect(x, y, w, 22, 4, COL_TOAST_BG);
  drawRoundRect(x, y, w, 22, 4, COL_DANGER);
  drawText5(x + 8, y + 7, gToastMsg, COL_TOAST_TEXT, COL_TOAST_BG, 1);
}
static void updateToast() {
  if (gToastUntil && (int32_t)(millis() - gToastUntil) >= 0) {
    gToastUntil = 0;
    paintScreen(gScreen);   /* repaint chrome + force a full value refresh */
  }
}

  switch (id) {
    case HOME_LIVE:  fillCircle(cx, cy, 7, color); break;

      break;
    case HOME_DEV:
      drawFastHLine(cx - 10, cy, 21, color);
      drawFastVLine(cx, cy - 10, 21, color);
      drawCircle(cx, cy, 4, color);
      break;
  }
}

    for (uint8_t i = 0; i < 2; i++) { energyCache[i][0] = 0; estCache[i][0] = 0; }

  drawFieldBig(LIVE_STAT[2].x + 4, LIVE_STAT[2].y + 38, 4, COL_TEMP, COL_BOX_FILL, val, statCache[2]);
  fmt3SigWatts(gState.milliwatts, val, sizeof(val));
  drawFieldBig(LIVE_STAT[3].x + 4, LIVE_STAT[3].y + 38, 4, COL_TEXT, COL_BOX_FILL, val, statCache[3]);

  snprintf(buf, sizeof(buf), "%.2f", gState.energyWh);

/* ==========================================================================
 * GRAPH — plot mechanics verbatim from v1.7 (autoscale, column-diff ink,
 * grid restore); span erase optimized to one windowed write per span.
 * ========================================================================*/
static bool gShowV = true, gShowT = true;
static int16_t gCurCeilingCentiamps = 1000;
static int16_t gVoltBaseLo, gVoltBaseHi;
static int16_t gVoltLo, gVoltHi;
static int16_t gTempHi = 5000;
#define TEMP_LO_CENTIDEGC 1500

static uint8_t gPrevY0[3][GRAPH_COLS], gPrevY1[3][GRAPH_COLS];

static const int16_t I_STEPS[] = { 2, 5, 10, 25, 50, 100, 150 };

static int16_t floorTo500(int32_t v) { int32_t q = v / 500; if (v < 0 && v % 500 != 0) q--; return (int16_t)(q * 500); }
static int16_t ceilTo500(int32_t v)  { int32_t q = v / 500; if (v > 0 && v % 500 != 0) q++; return (int16_t)(q * 500); }

static int16_t niceCeilAmps(int16_t maxCentiamps) {
  int32_t target = (int32_t)maxCentiamps * 115 / 100;
  for (uint8_t i = 0; i < 7; i++) if ((int32_t)I_STEPS[i] * 100 >= target) return I_STEPS[i] * 100;
  return I_STEPS[6] * 100;
}

static int16_t valueToY(int16_t value, int16_t lo, int16_t hi) {
  if (hi <= lo) return PLOT_Y1 - 1;
  float f = (float)(value - lo) / (float)(hi - lo);
  if (f < 0) f = 0;
  if (f > 1) f = 1;
  int16_t y = PLOT_Y1 - 2 - (int16_t)(f * (PLOT_H - 3));
  if (y < PLOT_Y0) y = PLOT_Y0;
  if (y > PLOT_Y1 - 2) y = PLOT_Y1 - 2;
  return y;
}

static bool isGridRow(int16_t y) {
  for (uint8_t r = 0; r < 6; r++) if (PLOT_Y0 + ((int32_t)r * PLOT_H) / 5 == y) return true;
  return false;
}

/* One windowed write per span (vs v1.7's per-pixel drawPixel loop) —
 * restores gridline pixels where the trace used to be. */
static void eraseSpan(int16_t x, int16_t y0, int16_t y1) {
  if (y0 < PLOT_Y0) y0 = PLOT_Y0;
  if (y1 > PLOT_Y1 - 1) y1 = PLOT_Y1 - 1;
  if (y1 < y0) return;
  uint16_t bgBe = __builtin_bswap16(COL_BG), grBe = __builtin_bswap16(COL_GRID);
  int16_t n = y1 - y0 + 1;
  for (int16_t i = 0; i < n; i++)
    s_lineBuf[i] = isGridRow(y0 + i) ? grBe : bgBe;
  spiBegin();
  lcdWindow(x, y0, 1, n);
  SPI.transfer((uint8_t *)s_lineBuf, nullptr, (size_t)n * 2);
  spiEnd();
}

static void drawGridFrame() {
  drawRect(PLOT_X0 - 1, PLOT_Y0 - 1, PLOT_W + 2, PLOT_H + 2, COL_BOX_BORDER);
  for (uint8_t r = 0; r < 6; r++) {
    int16_t y = PLOT_Y0 + ((int32_t)r * PLOT_H) / 5;
    drawFastHLine(PLOT_X0, y, PLOT_W, COL_GRID);
  }
  for (uint8_t c = 0; c < 6; c++) {
    int16_t x = PLOT_X0 + ((int32_t)c * PLOT_W) / 5;
    drawFastVLine(x, PLOT_Y0, PLOT_H, COL_GRID);
    char lbl[2]; snprintf(lbl, sizeof(lbl), "%d", 5 - c);
    drawText5(x - 2, PLOT_Y1 + 3, lbl, COL_TEXT_HI, COL_BG, 1);
  }
  /* uint16_t counter — a uint8_t against 380 hangs the MCU (v1.1 lesson) */
  for (uint16_t i = 0; i < GRAPH_COLS; i++) { gPrevY0[0][i] = gPrevY0[1][i] = gPrevY0[2][i] = 255; }
}

static void drawAxisLabels(bool forceClear) {
  static char iCache[6][8] = { "", "", "", "", "", "" };
  static char vCache[6][8] = { "", "", "", "", "", "" };
  static char tCache[6][8] = { "", "", "", "", "", "" };
  if (forceClear) for (uint8_t i = 0; i < 6; i++) { iCache[i][0] = vCache[i][0] = tCache[i][0] = 0; }
  char buf[8];
  int16_t vX = 444;
  int16_t tX = gShowV ? 420 : 444;
  for (uint8_t r = 0; r < 6; r++) {
    int16_t y = PLOT_Y0 + ((int32_t)r * PLOT_H) / 5 - 4;
    int16_t iVal = gCurCeilingCentiamps - (int32_t)r * gCurCeilingCentiamps / 5;
    fmtAxisAmps(iVal, buf, sizeof(buf));

    if (gShowV) {

      fmtWhole(tVal, buf, sizeof(buf));
      drawField5(tX, y, 4, 1, COL_TEMP, COL_BG, buf, tCache[r]);
    }
  }
}

/* Chip on/off must read at a glance: ON = channel-colored fill tint +

static void updateGraphTick(bool forceClear) {
  static char presCache[4][10] = { "", "", "", "" };
  static char peakCache[4][10] = { "", "", "", "" };
  if (forceClear) for (uint8_t i = 0; i < 4; i++) { presCache[i][0] = 0; peakCache[i][0] = 0; }
  char buf[16], val[10];

  for (uint8_t i = 0; i < 4; i++) {
    switch (i) {
      case 0: fmt3SigCenti(gState.centiamps, val, sizeof(val));  snprintf(buf, sizeof(buf), "%s %sA", GRAPH_PRESENT_LABEL[i], val); break;
      case 1: fmt3SigWatts(gState.milliwatts, val, sizeof(val)); snprintf(buf, sizeof(buf), "%s %sW", GRAPH_PRESENT_LABEL[i], val); break;
      case 2: fmt3SigCenti(gState.centivolts, val, sizeof(val)); snprintf(buf, sizeof(buf), "%s %sV", GRAPH_PRESENT_LABEL[i], val); break;
      case 3: fmtDegC(gState.centidegc, val, sizeof(val));       snprintf(buf, sizeof(buf), "%s %s",  GRAPH_PRESENT_LABEL[i], val); break;
    }
    const Rect &r = GRAPH_PRESENT[i];
    drawField5(r.x + 6, r.y + (r.h - 16) / 2, 7, 2, GRAPH_PRESENT_COLOR[i], COL_BOX_FILL, buf, presCache[i]);
  }

  for (uint8_t i = 0; i < 4; i++) {
    switch (i) {
      case 0: fmt3SigCenti(gState.peakCentiamps, val, sizeof(val));         snprintf(buf, sizeof(buf), "%s%sA", GRAPH_PEAK_LABEL[i], val); break;
      case 1: fmt3SigWatts(gState.peakMilliwatts, val, sizeof(val));        snprintf(buf, sizeof(buf), "%s%sW", GRAPH_PEAK_LABEL[i], val); break;
      case 2: fmt3SigCenti(gState.peakVAtPeakCentivolts, val, sizeof(val)); snprintf(buf, sizeof(buf), "%s%sV", GRAPH_PEAK_LABEL[i], val); break;
      case 3: fmtDegC(gState.peakCentidegc, val, sizeof(val));              snprintf(buf, sizeof(buf), "%s%s",  GRAPH_PEAK_LABEL[i], val); break;
    }
    const Rect &r = GRAPH_PEAK[i];
    drawField5(r.x + 6, r.y + (r.h - 16) / 2, 7, 2, GRAPH_PEAK_COLOR[i], COL_BOX_FILL, buf, peakCache[i]);
  }

  fmtTimer(gState.runElapsedMs, buf, sizeof(buf));
  drawField5(GRAPH_TIMER_BOX.x + 4, GRAPH_TIMER_BOX.y + 6, 10, 1, COL_TEXT, COL_BOX_FILL, buf, nullptr);

  /* Autoscale — pure integer scan (uint16_t counters, always). */
  int16_t maxI = 0, minV = 32767, maxV = -32768, maxT = TEMP_LO_CENTIDEGC;
  for (uint16_t c = 0; c < GRAPH_COLS; c++) {
    if (gRingI[c] > maxI) maxI = gRingI[c];
    if (gRingV[c] < minV) minV = gRingV[c];
    if (gRingV[c] > maxV) maxV = gRingV[c];
    if (gRingT[c] > maxT) maxT = gRingT[c];
  }
  gCurCeilingCentiamps = niceCeilAmps(maxI);
  int16_t candLo = floorTo500((int32_t)minV - 200);
  int16_t candHi = ceilTo500((int32_t)maxV + 200);
  gVoltLo = (candLo < gVoltBaseLo) ? candLo : gVoltBaseLo;
  gVoltHi = (candHi > gVoltBaseHi) ? candHi : gVoltBaseHi;
  int16_t candTHi = ceilTo500((int32_t)maxT + 300);
  gTempHi = (candTHi > 5000) ? candTHi : 5000;

  drawAxisLabels(forceClear);

  struct { const int16_t *ring; int16_t lo, hi; uint16_t color; bool show; } chans[3] = {
    { gRingI, 0, gCurCeilingCentiamps, COL_AMP, true },
    { gRingV, gVoltLo, gVoltHi, COL_VOLT, gShowV },
    { gRingT, TEMP_LO_CENTIDEGC, gTempHi, COL_TEMP, gShowT },
  };
  for (uint16_t c = 0; c < GRAPH_COLS; c++) {
    uint16_t idx = (gRingHead + 1 + c) % GRAPH_COLS;
    uint16_t prev = (idx + GRAPH_COLS - 1) % GRAPH_COLS;
    int16_t x = PLOT_X0 + c;
    for (uint8_t ch = 0; ch < 3; ch++) {
      if (!chans[ch].show) continue;
      int16_t yNow  = valueToY(chans[ch].ring[idx],  chans[ch].lo, chans[ch].hi);
      int16_t yPrev = (c == 0) ? yNow : valueToY(chans[ch].ring[prev], chans[ch].lo, chans[ch].hi);
      int16_t y0 = (yNow < yPrev) ? yNow : yPrev;
      int16_t y1 = ((yNow > yPrev) ? yNow : yPrev) + 1;
      if (gPrevY0[ch][c] != 255 && (gPrevY0[ch][c] != y0 || gPrevY1[ch][c] != y1))
        eraseSpan(x, gPrevY0[ch][c], gPrevY1[ch][c]);
      drawFastVLine(x, y0, y1 - y0 + 1, chans[ch].color);
      gPrevY0[ch][c] = (uint8_t)y0; gPrevY1[ch][c] = (uint8_t)y1;
    }
  }
}

  if (sn.down)
    for (uint8_t i = 0; i < 4; i++)
      if (inRect(DEV_TARGETS[i].t->hit, mx, my)) { hit = DEV_TARGETS[i].label; break; }
  snprintf(buf, sizeof(buf), "HIT: %s", hit);
  drawField5(X, 112, 19, 1, COL_TEXT_HI, COL_BG, buf, cache[3]);

  uint32_t hz = gLoopUsAvg ? (1000000UL / gLoopUsAvg) : 0;
  snprintf(buf, sizeof(buf), "TP %luHZ  UI %luHZ", (unsigned long)g_touchHz, (unsigned long)hz);
  drawField5(X, 128, 20, 1, COL_TEXT_HI, COL_BG, buf, cache[4]);
  snprintf(buf, sizeof(buf), "DN %lu UP %lu TAP %lu", (unsigned long)g_cntDown,
           (unsigned long)g_cntUp, (unsigned long)gDevTaps);
  drawField5(X, 144, 26, 1, COL_TEXT_HI, COL_BG, buf, cache[5]);
  snprintf(buf, sizeof(buf), "INT %lu  I2C ERR %lu", (unsigned long)g_intEdges,
           (unsigned long)g_i2cErr);
  drawField5(X, 160, 24, 1, COL_TEXT_HI, COL_BG, buf, cache[6]);
  snprintf(buf, sizeof(buf), "G RNG %lu JMP %lu DRP %lu OVF %lu", (unsigned long)g_glRange,
           (unsigned long)g_glJump, (unsigned long)g_glDrop, (unsigned long)g_glOvf);
  drawField5(X, 176, 34, 1, COL_TEXT_HI, COL_BG, buf, cache[7]);

  /* Bold crosshair: erase old footprint in bg (slivers through static
   * chrome are a known, self-contained cosmetic artifact on this debug
   * screen — cleared on re-entry), then draw at the new position. */
  if (wasDown && (!sn.down || lastCx != mx || lastCy != my))
    drawDevCross(lastCx, lastCy, COL_BG);
  if (sn.down) {
    drawDevCross(mx, my, COL_AMP);
    lastCx = mx; lastCy = my;
  }
  wasDown = sn.down;
}

static void devSetPressed(int8_t id, bool pressed) { (void)id; drawBackBtn(DEV_T[0].vis, pressed); }
static void devDispatch(int8_t id) { (void)id; goTo(SCR_HOME); }

/* ==========================================================================
 * Screen dispatch + navigation
 * ========================================================================*/
static void paintScreen(ScreenId s) {
  switch (s) {
    case SCR_HOME:  paintHomeOnce();  break;
    case SCR_LIVE:  paintLiveOnce();  break;
    case SCR_GRAPH: paintGraphOnce(); break;
    case SCR_CAL:   paintCalOnce();   break;
    case SCR_DEV:   paintDevOnce();   break;
  }
}
static void goTo(ScreenId s) { gScreen = s; paintScreen(s); }

static int8_t hitTestScreen(ScreenId s, int16_t x, int16_t y) {
  switch (s) {
    case SCR_HOME:  return hitTestTargets(HOME_T, HOME_N, x, y);
    case SCR_LIVE:  return hitTestTargets(LIVE_T, LIVE_BTN_N, x, y);
    case SCR_GRAPH: return hitTestTargets(GRAPH_T, GRAPH_BTN_N, x, y);
    case SCR_CAL:   return hitTestTargets(CAL_T, 1, x, y);
    case SCR_DEV:   return hitTestTargets(DEV_T, 1, x, y);
  }
  return -1;
}
static void dispatch(ScreenId s, int8_t id) {
  switch (s) {
    case SCR_HOME:  homeDispatch(id);  break;
    case SCR_LIVE:  liveDispatch(id);  break;
    case SCR_GRAPH: graphDispatch(id); break;
    case SCR_CAL:   calDispatch(id);   break;
    case SCR_DEV:   devDispatch(id);   break;
  }
}
static void setPressedVisual(ScreenId s, int8_t id, bool pressed) {
  switch (s) {
    case SCR_HOME:  homeSetPressed(id, pressed);  break;
    case SCR_LIVE:  liveSetPressed(id, pressed);  break;
    case SCR_GRAPH: graphSetPressed(id, pressed); break;
    case SCR_CAL:   calSetPressed(id, pressed);   break;
    case SCR_DEV:   devSetPressed(id, pressed);   break;
  }
}

/* ==========================================================================
 * Touch event pump — core 0. Drains the inter-core FIFO every loop pass.
 * Dispatch fires on the PRESS edge (v1.7 semantics — feels snappy).
 * gPressedScreen guards against a stale pressed-visual clear after a
 * dispatch navigated away (a latent v1.7 quirk, fixed here).
 * ========================================================================*/
static int8_t   gPressedId = -1;
static ScreenId gPressedScreen = SCR_HOME;
static int16_t  gDownX = 0, gDownY = 0;
static uint32_t gDownMs = 0;

static void pumpTouchEvents() {
  uint32_t ev;
  while (rp2040.fifo.pop_nb(&ev)) {
    uint32_t type = ev >> 28;
    int16_t mx, my;
    mapTouch((ev >> 12) & 0xFFF, ev & 0xFFF, mx, my);

    if (type == EV_DOWN) {
      gDownX = mx; gDownY = my; gDownMs = millis();
      int8_t id = hitTestScreen(gScreen, mx, my);
      if (id >= 0) {
        gPressedId = id;
        gPressedScreen = gScreen;
        setPressedVisual(gScreen, id, true);
        dispatch(gScreen, id);
      }
    } else if (type == EV_UP) {
      int16_t dx = mx - gDownX, dy = my - gDownY;
      if (dx < 0) dx = -dx;
      if (dy < 0) dy = -dy;
      if (millis() - gDownMs < TAP_MS && dx < TAP_MOVE_PX && dy < TAP_MOVE_PX) gDevTaps++;
      if (gPressedId >= 0) {
        if (gScreen == gPressedScreen) setPressedVisual(gScreen, gPressedId, false);
        gPressedId = -1;
      }
    }
  }
}

/* ==========================================================================
 * setup() / loop() — core 0
 * ========================================================================*/
#define LIVE_FRAME_MS   50
#define GRAPH_FRAME_MS  33
#define DEV_FRAME_MS    50

void setup() {
  /* SD_CS must never float low — first GPIO touched (shares FPC's MISO). */
  pinMode(PIN_SD_CS, OUTPUT); digitalWrite(PIN_SD_CS, HIGH);
  pinMode(PIN_LCD_LED, OUTPUT); digitalWrite(PIN_LCD_LED, HIGH);  /* solid, never PWM */

  analogReadResolution(12);
  Serial.begin(115200);

  lcdBegin();
  digitalWrite(PIN_LCD_LED, HIGH);

  Serial.println(F("Logging Current Meter UI  v2.1 (FABLE)"));
  checkTargetOverlaps();

  goTo(SCR_HOME);
  gCoreReady = true;   /* release core 1 last (it holds touch; ADC starts now) */
}

void loop() {
  static uint32_t lastLoopUs = 0;
  uint32_t nowUs = micros();
  if (lastLoopUs) {
    uint32_t dtUs = nowUs - lastLoopUs;
    gLoopUsAvg = gLoopUsAvg ? (gLoopUsAvg + ((int32_t)dtUs - (int32_t)gLoopUsAvg) / 16) : dtUs;
  }
  lastLoopUs = nowUs;

  pumpTouchEvents();   /* never skipped, even on a slow frame */

  static uint32_t lastLiveFrame = 0, lastGraphFrame = 0, lastDevFrame = 0;
  uint32_t now = millis();
  switch (gScreen) {
    case SCR_LIVE:
      if (now - lastLiveFrame >= LIVE_FRAME_MS) { lastLiveFrame = now; updateLiveTick(); }
      break;
    case SCR_GRAPH:
      if (now - lastGraphFrame >= GRAPH_FRAME_MS) {
        lastGraphFrame = now;
        updateGraphTick(gGraphForceRedraw);   /* forced after a V/T toggle */
        gGraphForceRedraw = false;
      }
      break;
    case SCR_DEV:
      if (now - lastDevFrame >= DEV_FRAME_MS) { lastDevFrame = now; updateDevTick(); }
      break;
    default: break;
  }

  updateToast();

  if (Serial.available()) {
    char c = Serial.read();
    if (c == 'r' || c == 'R') {
      g_statReset = true;
      gDevTaps = 0;
      Serial.println(F("[touch stats reset]"));
    }
  }
}
